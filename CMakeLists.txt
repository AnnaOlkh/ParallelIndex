cmake_minimum_required(VERSION 3.20)
project(parallel_index LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
option(ENABLE_UNSAFE_V1 "Build the intentionally undefined-behavior V1 mode" OFF)
add_library(index_core src/core.cpp src/versions.cpp)
target_include_directories(index_core PUBLIC src)
target_link_libraries(index_core PUBLIC Threads::Threads)
target_compile_definitions(index_core PUBLIC LAB_UNSAFE_V1=$<BOOL:${ENABLE_UNSAFE_V1}>)
add_executable(indexer src/main.cpp)
target_link_libraries(indexer PRIVATE index_core)
add_executable(false_sharing src/false_sharing.cpp)
target_link_libraries(false_sharing PRIVATE Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(index_tests tests/tests.cpp)
  target_link_libraries(index_tests PRIVATE index_core)
  target_compile_definitions(index_tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME index_tests COMMAND index_tests)
endif()
foreach(target index_core indexer false_sharing)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(BUILD_TESTING)
  if(MSVC)
    target_compile_options(index_tests PRIVATE /W4 /utf-8)
  else()
    target_compile_options(index_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
